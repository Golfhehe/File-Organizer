    d = d or date.today()

        self.current_date = date(qdate.year(), qdate.month(), qdate.day())

            item = self.images.takeAt(0)

        if not path.exists():
            return
        label = QLabel()
        label.setFixedSize(180, 130)
        label.setAlignment(Qt.AlignmentFlag.AlignCenter)
        label.setToolTip(str(path))
        pix = QPixmap(str(path))
        if not pix.isNull():
            label.setPixmap(

                    170, 120,
                    Qt.AspectRatioMode.KeepAspectRatio,
                    Qt.TransformationMode.SmoothTransformation

class PomodoroPage(QWidget):
    def __init__(self, main):
        super().__init__()
        self.main = main
        self.running = False
        self.remaining = 25 * 60
        self.mode = "Học"

        root = QVBoxLayout(self)
        root.setAlignment(Qt.AlignmentFlag.AlignCenter)

        self.mode_label = QLabel("🍅 POMODORO — HỌC")
        self.mode_label.setObjectName("section")
        self.clock = QLabel("25:00")
        self.clock.setObjectName("pomodoro")
        self.status = QLabel("Sẵn sàng. Khi bắt đầu, app sẽ nhắc ông bật app chống mất tập trung.")
        self.status.setWordWrap(True)
        self.status.setAlignment(Qt.AlignmentFlag.AlignCenter)

        root.addWidget(self.mode_label, alignment=Qt.AlignmentFlag.AlignCenter)
        root.addWidget(self.clock, alignment=Qt.AlignmentFlag.AlignCenter)
        root.addWidget(self.status, alignment=Qt.AlignmentFlag.AlignCenter)

        row = QHBoxLayout()
        start = QPushButton("▶ Bắt đầu")
        pause = QPushButton("⏸ Tạm dừng")
        reset = QPushButton("↺ Đặt lại")
        start.clicked.connect(self.start)
        pause.clicked.connect(self.pause)
        reset.clicked.connect(self.reset)
        row.addWidget(start)
        row.addWidget(pause)
        row.addWidget(reset)
        root.addLayout(row)

        presets = QHBoxLayout()
        for text, sec in [("25 / 5", 25*60), ("50 / 10", 50*60), ("90 / 15", 90*60)]:
            b = QPushButton(text)
            b.clicked.connect(lambda _, s=sec: self.set_work_time(s))
            presets.addWidget(b)
        root.addLayout(presets)

        self.timer = QTimer(self)
        self.timer.timeout.connect(self.tick)

    def set_work_time(self, seconds):
        if not self.running:
            self.remaining = seconds
            self.clock.setText(self.format_time())
            self.status.setText("Đã chọn phiên học mới.")

    def format_time(self):
        return f"{self.remaining // 60:02d}:{self.remaining % 60:02d}"

    def start(self):
        if self.running:
            return
        self.running = True
        self.timer.start(1000)
        self.status.setText("Tập trung thôi ông. Bật app chống mất tập trung nếu cần nhé.")
        if self.main.data["settings"].get("focus_reminder", True):
            self.main.notify(
                "🍅 Pomodoro bắt đầu",
                "Bật app chống mất tập trung, đóng tab linh tinh và chiến thôi =))"
            )

    def pause(self):
        self.running = False
        self.timer.stop()
        self.status.setText("Đã tạm dừng.")

    def reset(self):
        self.running = False
        self.timer.stop()
        self.remaining = 25 * 60
        self.mode = "Học"
        self.mode_label.setText("🍅 POMODORO — HỌC")
        self.clock.setText(self.format_time())
        self.status.setText("Sẵn sàng.")

    def tick(self):
        self.remaining -= 1
        self.clock.setText(self.format_time())
        if self.remaining <= 0:
            self.running = False
            self.timer.stop()
            if self.mode == "Học":
                self.main.add_study_minutes(25, silent=True)
                self.main.notify("🎉 Hết phiên học!", "Ông vừa hoàn thành một phiên Pomodoro. Nghỉ 5 phút nhé.")
                self.mode = "Nghỉ"
                self.mode_label.setText("☕ POMODORO — NGHỈ")
                self.remaining = 5 * 60
                self.clock.setText(self.format_time())
                self.status.setText("Nghỉ 5 phút. Sau đó có thể bắt đầu phiên tiếp theo.")
            else:
                self.notify_break_done()

    def notify_break_done(self):
        self.main.notify("⏰ Hết giờ nghỉ", "Quay lại học thôi ông =))")
        self.mode = "Học"
        self.mode_label.setText("🍅 POMODORO — HỌC")
        self.remaining = 25 * 60
        self.clock.setText(self.format_time())
        self.status.setText("Sẵn sàng cho phiên học tiếp theo.")

class AIPage(QWidget):
    def __init__(self, main):
        super().__init__()
        self.main = main
        root = QVBoxLayout(self)

        title = QLabel("🤖 TRỢ LÝ 300 NGÀY")
        title.setObjectName("title_small")
        root.addWidget(title)

        self.analysis = QLabel()
        self.analysis.setWordWrap(True)
        self.analysis.setObjectName("ai_box")
        root.addWidget(self.analysis)

        self.plan = QTextEdit()
        self.plan.setReadOnly(True)
        root.addWidget(QLabel("📋 Kế hoạch đề xuất"))
        root.addWidget(self.plan, 1)

        b = QPushButton("🤖 Tạo kế hoạch hôm nay")
        b.clicked.connect(self.generate)
        root.addWidget(b)

        self.generate()

    def generate(self):
        d = self.main.data
        today = today_key()
        h = d["history"].get(today, {})
        minutes = h.get("minutes", 0)
        tasks_done = h.get("tasks", 0)

        by_subject = {}
        for t in d["tasks"]:
            if t["done"]:
                by_subject[t["subject"]] = by_subject.get(t["subject"], 0) + 1

        subject_totals = {}
        for t in d["tasks"]:
            subject_totals[t["subject"]] = subject_totals.get(t["subject"], 0) + 1

        weakest = None
        weakest_pct = 101
        for sub, total in subject_totals.items():
            pct = by_subject.get(sub, 0) / total * 100
            if pct < weakest_pct:
                weakest_pct = pct
                weakest = sub

        n = day_number()
        phase, goal = phase_for_day(n)
        left = max(0, (EXAM_DATE - date.today()).days)

        lines = [
            f"📅 Hôm nay là ngày {n}/{TOTAL_DAYS}. Còn {left} ngày.",
            f"🧭 Giai đoạn: {phase} — {goal}.",
            f"⏱ Hôm nay đã ghi nhận: {minutes} phút học.",
            f"✓ Nhiệm vụ hoàn thành hôm nay: {tasks_done}.",
        ]

        if minutes < d["settings"].get("daily_goal", 120):
            lines.append(
                f"⚠️ Ông đang dưới mục tiêu {d['settings'].get('daily_goal',120)} phút."
            )
        else:
            lines.append("🔥 Mục tiêu thời gian hôm nay đã đạt.")

        if weakest:
            lines.append(
                f"📚 Môn đang cần chú ý: {weakest} ({weakest_pct:.0f}% nhiệm vụ hiện tại đã hoàn thành)."
            )

        self.analysis.setText("\n".join(lines))

        plan = []
        if weakest:
            plan.append(f"1. 📚 {weakest} — 45 phút\n   Ưu tiên phần đang yếu nhất.")
        if "Tin" != weakest:
            plan.append("2. 💻 Tin/C++ — 30 phút\n   Luyện 1–2 bài ngắn, giữ nhịp.")
        plan.append("3. 🧠 Ôn lại lỗi hôm nay — 15 phút\n   Ghi vào nhật ký học tập.")
        plan.append(f"\nTổng đề xuất: khoảng {45 + 30 + 15} phút.")
        self.plan.setPlainText("\n\n".join(plan))

        title.setObjectName("title_small")
        root.addWidget(title)

        self.tasks = QListWidget()
        self.tasks.itemDoubleClicked.connect(self.toggle_task)
        root.addWidget(self.tasks, 1)

        row = QHBoxLayout()
        for text, fn in [
            ("+ Thêm", self.add_task),
            ("✓ Hoàn thành", self.toggle_selected),

        ]:

            b.clicked.connect(fn)
            row.addWidget(b)

        root.addWidget(self.task_progress)
        self.task_progress_text = QLabel()
        root.addWidget(self.task_progress_text)
        return w

    def refresh_all(self):
        self.refresh_tasks()
        self.dashboard.refresh()
        self.ai_page.generate()
        self.journal.load_selected_day(
            QDate(self.journal.current_date.year,
                  self.journal.current_date.month,
                  self.journal.current_date.day)
        )

    def refresh_tasks(self):
        self.tasks.clear()
        done = 0
        for i, task in enumerate(self.data["tasks"]):
            mark = "✓" if task["done"] else "○"
            item = QListWidgetItem(
                f"{mark}  [{task['subject']}]  {task['name']}"
            )
            item.setData(Qt.ItemDataRole.UserRole, i)
            self.tasks.addItem(item)
            if task["done"]:
                done += 1
        total = len(self.data["tasks"])
        pct = round(done / total * 100) if total else 0
        self.task_progress.setValue(pct)
        self.task_progress_text.setText(
            f"{done}/{total} nhiệm vụ hoàn thành • {pct}%"
        )

    def selected_index(self):
        item = self.tasks.currentItem()
        return item.data(Qt.ItemDataRole.UserRole) if item else None

    def add_task(self):
        d = AddTaskDialog(self)
        if d.exec():
            name = d.name.text().strip()
            if not name:
                QMessageBox.warning(self, "Thiếu nhiệm vụ", "Nhập nhiệm vụ trước nha.")
                return
            self.data["tasks"].append({
                "name": name,
                "subject": d.subject.currentText(),
                "done": False,
                "xp": 20
            })
            save_data(self.data)
            self.refresh_all()

    def toggle_task(self, item):
        self._toggle(item.data(Qt.ItemDataRole.UserRole))

    def toggle_selected(self):
        i = self.selected_index()
        if i is not None:
            self._toggle(i)

    def _toggle(self, i):
        was_done = self.data["tasks"][i]["done"]
        self.data["tasks"][i]["done"] = not was_done
        if not was_done:
            self.data["xp"] += self.data["tasks"][i].get("xp", 20)
            self.record_task_done()
            self.notify("✓ Nhiệm vụ hoàn thành", "+20 XP. Tiếp tục chiến!")
        save_data(self.data)
        self.refresh_all()

    def record_task_done(self):
        key = today_key()
        h = self.data["history"].setdefault(
            key, {"minutes": 0, "tasks": 0, "note": "", "images": []}
        )
        h["tasks"] += 1

    def delete_task(self):
        i = self.selected_index()
        if i is not None:
            del self.data["tasks"][i]
            save_data(self.data)
            self.refresh_all()

    def add_study_minutes(self, minutes, silent=False):
        self.data["study_minutes"] += minutes
        key = today_key()
        h = self.data["history"].setdefault(
            key, {"minutes": 0, "tasks": 0, "note": "", "images": []}
        )
        h["minutes"] += minutes

        last = self.data.get("last_study")
        today = date.today()
        if last != str(today):
            if last == str(today - timedelta(days=1)):
                self.data["streak"] += 1
            else:
                self.data["streak"] = 1
            self.data["last_study"] = str(today)

        self.data["xp"] += max(1, minutes // 3)
        save_data(self.data)
        self.refresh_all()
        if not silent:
            self.notify("⏱ Đã ghi thời gian", f"+{minutes} phút học • XP tăng.")

        spin = QSpinBox()

        spin.setValue(30)
        spin.setSuffix(" phút")
        form.addRow("Đã học:", spin)
        buttons = QDialogButtonBox(

        )
        buttons.accepted.connect(d.accept)
        buttons.rejected.connect(d.reject)
        form.addRow(buttons)
        if d.exec():
            self.add_study_minutes(spin.value())

    def notify(self, title, message):
        # Cross-platform fallback: dialog; Windows tray notifications can be
        # added later without changing the data model.
        if not self.isActiveWindow():
            QMessageBox.information(self, title, message)
        else:
            # Small status popup is less intrusive while user is in the app.
            self.status_message = QLabel(f"{title}: {message}", self)
            self.status_message.setObjectName("toast")
            self.status_message.setWordWrap(True)
            self.status_message.adjustSize()
            self.status_message.move(30, self.height() - 100)
            self.status_message.show()
            QTimer.singleShot(4500, self.status_message.deleteLater)

    def background_check(self):
        now = datetime.now()
        if (
            self.data["settings"].get("end_day_reminder", True)
            and now.strftime("%H:%M") == self.data["settings"].get("end_day_time", "21:30")
            and self.last_end_day_notification != today_key()
        ):
            self.last_end_day_notification = today_key()
            h = self.data["history"].get(today_key(), {})
            mins = h.get("minutes", 0)
            self.notify(
                "🌙 Cuối ngày rồi!",
                f"Ông đã ghi {mins} phút hôm nay. Nhớ lưu nhật ký và tiến độ trước khi nghỉ nhé."
            )

    def closeEvent(self, event):
        # Always remind before closing, but allow quick exit.
        reply = QMessageBox.question(
            self,
            "Lưu tiến độ",
            "Trước khi thoát: ông đã lưu nhật ký hôm nay chưa?\n"
            "Dữ liệu tự lưu khi thao tác, nhưng nhật ký chữ cần bấm 'Lưu nhật ký'.",
            QMessageBox.StandardButton.Yes | QMessageBox.StandardButton.No
        )
        if reply == QMessageBox.StandardButton.Yes:
            event.accept()
        else:
            event.ignore()
